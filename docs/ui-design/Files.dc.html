36import bisect
from pathlib import Path

import torch
from cachetools import LRUCache
from torch.utils.data import Dataset

from .index import read_offsets


class ShardedTextDataset(Dataset):
    """按分片读取 token，只缓存最近使用的分片。"""

    def __init__(self, shards, tokenizer, seq_len=2048, cache_shards=4):
        self.shards = [Path(p) for p in shards]
        self.tokenizer = tokenizer
        self.seq_len = seq_len
        # 每个 worker 最多保留 cache_shards 个分片，内存不再随 epoch 增长
        self._cache = LRUCache(maxsize=cache_shards)
        self._cache_shards = cache_shards
        self._offsets = read_offsets(self.shards, seq_len)

    def __len__(self):
        return self._offsets[-1]

    def __getitem__(self, idx):
        shard_id = self._shard_of(idx)
        shard = self._cache.get(shard_id)
        if shard is None:
            shard = torch.load(self.shards[shard_id], mmap=True)
            self._cache[shard_id] = shard
        start = (idx - self._offsets[shard_id]) * self.seq_len
        return shard[start : start + self.seq_len]

    def _shard_of(self, idx):
        return bisect.bisect_right(self._offsets, idx) - 1

25import argparse

import yaml
from torch.utils.data import DataLoader

from data.loader import ShardedTextDataset
from models import build_model, load_tokenizer


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--config", default="configs/train.yaml")
    args = parser.parse_args()
    with open(args.config) as f:
        cfg = yaml.safe_load(f)

    tokenizer = load_tokenizer(cfg["tokenizer"])
    dataset = ShardedTextDataset(tokenizer=tokenizer, **cfg["data"])
    loader = DataLoader(dataset, batch_size=cfg["batch_size"], num_workers=4)
    model = build_model(cfg["model"])
    model.fit(loader, epochs=cfg["epochs"])


if __name__ == "__main__":
    main()

18# llm-train

基于 PyTorch 的语言模型训练代码。训练数据按分片存放在 `/data/shards`，
超参数集中在 `configs/`。

## 快速开始

```bash
pip install -e .
python -m train --config configs/train.yaml
```

## 目录结构

- `src/data/`：分片数据集，按分片 LRU 缓存
- `src/models/`：模型定义与 tokenizer
- `configs/`：训练与评测配置
- `tests/`：单元测试，运行 `pytest tests/ -q`
